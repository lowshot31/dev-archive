import { LightningElement } from 'lwc';
import { ShowToastEvent } from 'lightning/platformShowToastEvent';

export default class GuestInquiryForm extends LightningElement {
    daum = null;
    daumLoading = false;

    /**
     * Experience Cloud에서 카카오 API를 안전하게 로드
     */
    loadDaumAPI() {
        return new Promise((resolve, reject) => {
            if (this.daum) {
                resolve(this.daum);
                return;
            }

            if (this.daumLoading) {
                // 이미 로드 중이면 기다리기
                setTimeout(() => this.loadDaumAPI().then(resolve).catch(reject), 100);
                return;
            }

            this.daumLoading = true;

            const script = document.createElement('script');
            script.src = 'https://t1.daumcdn.net/mapjsapi/bundle/postcode/prod/postcode.v2.js';
            script.async = true;
            script.onload = () => {
                this.daum = window.daum;
                this.daumLoading = false;
                resolve(this.daum);
            };
            script.onerror = () => {
                this.daumLoading = false;
                reject(new Error('카카오 API 로드 실패'));
            };
            document.head.appendChild(script);
        });
    }

    /**
     * 카카오 우편주소 검색 핸들러
     */
    async handleAddressSearch() {
        try {
            await this.loadDaumAPI();

            if (!window.daum || !window.daum.Postcode) {
                throw new Error('우편번호 서비스가 준비되지 않았습니다.');
            }

            new daum.Postcode({
                oncomplete: (data) => {
                    this.parseAddressData(data);
                },
                onclose: (state) => {
                    console.log('우편번호 팝업 닫기:', state);
                },
                onerror: (errorCode) => {
                    console.error('우편번호 API 에러:', errorCode);
                }
            }).open();
        } catch (error) {
            const evt = new ShowToastEvent({
                title: '오류',
                message: error.message || '우편번호 서비스를 로드할 수 없습니다.',
                variant: 'error',
            });
            this.dispatchEvent(evt);
        }
    }

    /**
     * 카카오 API에서 받은 주소 데이터 파싱
     * @param {Object} data - 카카오 API 반환 데이터
     */
    parseAddressData(data) {
        try {
            // 데이터 확인
            const roadAddress = data.roadAddress || data.address || '';
            
            // sido = 광역시/도, sigungu = 시/군/구
            let region = data.sido || ''; // 예: "서울"
            let district = data.sigungu || ''; // 예: "강남구"

            // 더 정확한 파싱 (shorthand가 false인 경우도 처리)
            if (!region && data.address) {
                const parts = data.address.split(' ');
                region = parts[0] || '';
                district = parts[1] || '';
            }

            console.log('주소 파싱 완료:', {
                roadAddress,
                region,
                district,
                fullData: data
            });

            this.updateFields(roadAddress, region, district);
        } catch (error) {
            const evt = new ShowToastEvent({
                title: '오류',
                message: '주소 파싱 중 오류가 발생했습니다.',
                variant: 'error',
            });
            this.dispatchEvent(evt);
        }
    }

    /**
     * 폼 필드 업데이트
     * @param {string} address - 도로명 주소
     * @param {string} region - 지역 (광역시/도)
     * @param {string} district - 세부지역 (시/군/구)
     */
    updateFields(address, region, district) {
        try {
            const companyInput = this.template.querySelector('lightning-input-field[field-name="Company"]');
            const regionInput = this.template.querySelector('lightning-input-field[field-name="Region__c"]');
            const districtInput = this.template.querySelector('lightning-input-field[field-name="District__c"]');

            if (companyInput && address) {
                companyInput.value = address;
            }
            if (regionInput && region) {
                regionInput.value = region;
            }
            if (districtInput && district) {
                districtInput.value = district;
            }

            const evt = new ShowToastEvent({
                title: '성공',
                message: `주소가 입력되었습니다.\n지역: ${region} | 세부지역: ${district}`,
                variant: 'success',
            });
            this.dispatchEvent(evt);
        } catch (error) {
            console.error('필드 업데이트 오류:', error);
        }
    }

    handleSuccess(event) {
        const evt = new ShowToastEvent({
            title: '성공',
            message: '견적 문의가 성공적으로 접수되었습니다!',
            variant: 'success',
        });
        this.dispatchEvent(evt);
    }

    handleError(event) {
        console.error('폼 저장 오류:', event.detail);
        const evt = new ShowToastEvent({
            title: '오류 발생',
            message: '접수 중 문제가 발생했습니다. 관리자에게 문의하세요.',
            variant: 'error',
        });
        this.dispatchEvent(evt);
    }
}
